import {
  BrowserRouter,
  Navigate,
  Route,
  Routes,
} from "react-router-dom";

import Login from "./pages/Login";
import Dashboard from "./pages/Dashboard";
import NewProcedure from "./pages/NewProcedure";
import RFIDRegistration from "./pages/RFIDRegistration";
import PreOpScan from "./pages/PreOpScan";
import ProcedureActive from "./pages/ProcedureActive";
import PostOpScan from "./pages/PostOpScan";
import FinalReconciliation from "./pages/FinalReconciliation";

export default function App() {
  return (
    <BrowserRouter>

      <Routes>

        <Route
          path="/"
          element={
            <Login />
          }
        />

        <Route
          path="/dashboard"
          element={
            <Dashboard />
          }
        />

        <Route
          path="/new-procedure"
          element={
            <NewProcedure />
          }
        />

        <Route
          path="/rfid-registration"
          element={
            <RFIDRegistration />
          }
        />

        <Route
          path="/pre-op-scan"
          element={
            <PreOpScan />
          }
        />

        <Route
          path="/procedure-active"
          element={
            <ProcedureActive />
          }
        />

        <Route
          path="/post-op-scan"
          element={
            <PostOpScan />
          }
        />

        <Route
          path="/final-reconciliation"
          element={
            <FinalReconciliation />
          }
        />

        <Route
          path="*"
          element={
            <Navigate
              to="/"
              replace
            />
          }
        />

      </Routes>

    </BrowserRouter>
  );
}

export const API_BASE_URL = (
  import.meta.env.VITE_API_BASE_URL ||
  "http://127.0.0.1:8000"
).replace(/\/$/, "");


/* =========================================================
   GENERIC REQUEST HELPER
========================================================= */

async function apiRequest(
  path,
  options = {}
) {
  const {
    method = "GET",
    body = null,
    auth = false,
    headers = {},
  } = options;

  const requestHeaders = {
    Accept: "application/json",
    ...headers,
  };

  /*
   * Automatically attach JSON content type
   * for normal JSON requests.
   */

  if (
    body !== null &&
    !(body instanceof FormData)
  ) {
    requestHeaders["Content-Type"] =
      "application/json";
  }

  /*
   * Attach JWT when endpoint requires
   * authentication.
   */

  if (auth) {
    const token =
      getAuthToken();

    if (!token) {
      throw new Error(
        "Authentication required."
      );
    }

    requestHeaders.Authorization =
      `Bearer ${token}`;
  }

  let response;

  try {
    response = await fetch(
      `${API_BASE_URL}${path}`,
      {
        method,

        headers:
          requestHeaders,

        body:
          body === null
            ? undefined
            : body instanceof FormData
            ? body
            : JSON.stringify(
                body
              ),
      }
    );
  } catch (error) {
    console.error(
      "API network error:",
      error
    );

    throw new Error(
      "Unable to connect to the SurgiTrack server. Make sure the backend is running."
    );
  }

  /*
   * Try to read backend response.
   */

  let data = null;

  const contentType =
    response.headers.get(
      "content-type"
    ) || "";

  try {
    if (
      contentType.includes(
        "application/json"
      )
    ) {
      data =
        await response.json();
    } else {
      const text =
        await response.text();

      data =
        text || null;
    }
  } catch {
    data = null;
  }

  /*
   * FastAPI normally returns:
   *
   * {
   *   "detail": "Error message"
   * }
   */

  if (!response.ok) {
    let message =
      "Request failed.";

    if (
      data &&
      typeof data === "object" &&
      data.detail
    ) {
      if (
        typeof data.detail ===
        "string"
      ) {
        message =
          data.detail;
      } else {
        message =
          "The server rejected the request.";
      }
    } else if (
      typeof data === "string" &&
      data
    ) {
      message =
        data;
    }

    throw new Error(
      message
    );
  }

  return data;
}

    []
  );

export function addRfidScan(
  scan
) {
  const current =
    getRfidScans();

  const duplicate =
    current.some(
      (item) =>
        item.uid ===
        scan.uid
    );

  if (duplicate) {
    return {
      added: false,
      scans: current,
    };
  }

  const updated = [
    ...current,
    scan,
  ];

  write(
    employeeKey(
      "rfid_scans"
    ),
    updated
  );

  return {
    added: true,
    scans: updated,
  };
}


export function removeRfidScan(
  uid
) {
  const updated =
    getRfidScans().filter(
      (item) =>
        item.uid !== uid
    );

  write(
    employeeKey(
      "rfid_scans"
    ),
    updated
  );

  return updated;
}


export function clearRfidScans() {

  );
}

/* =========================================================
   ARCHIVE PROCEDURE
========================================================= */

export function archiveCurrentProcedure() {
  const user =
    getCurrentUser();

  const procedure =
    getCurrentProcedure();

  const scans =
    getRfidScans();

  const vision =
    getVisionStatus();

  if (!procedure) {
    return null;
  }

  const completedAt =
    new Date().toISOString();

  const archiveRecord = {
    historyId:
      `SURGI-${Date.now()}`,

    caseId:
      procedure.caseId,

    procedureName:
      procedure.procedureName,

    otRoom:
      procedure.otRoom,

    surgeon:
      procedure.surgeon,

    startedAt:
      procedure.createdAt,

    completedAt,

    status:
      "Completed",

    attendedBy: {
      name:
        user.name,

      employeeId:
        user.employeeId,

      role:
        user.role,
    },

    rfidScans:
      scans.map(
        (scan) => ({
          ...scan,
        })
      ),

    instrumentCount:
      scans.length,

    vision: {
      preOpCaptured:
        Boolean(
          vision.preOpCaptured
        ),

      postOpCaptured:
        Boolean(
          vision.postOpCaptured
        ),

      preOpCapturedAt:
        vision.preOpCapturedAt ||
        null,

      postOpCapturedAt:
        vision.postOpCapturedAt ||
        null,
    },

    visionReconciliation:
      "Pending",
  };

  const history =
    getProcedureHistory();

  const updatedHistory = [
    archiveRecord,
    ...history,
  ];

  saveProcedureHistory(
    updatedHistory
  );

  /*
   * Move completed workflow out of active state.
   */

  clearActiveProcedureState();

  return archiveRecord;
}


/* =========================================================
   CLEAR ACTIVE WORKFLOW

   User history remains untouched.
========================================================= */

export function clearActiveProcedureState() {
  clearCurrentProcedure();

  clearRfidScans();

  clearVisionStatus();
}
